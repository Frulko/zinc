cmake_minimum_required(VERSION 3.20)
project(carplay_usb LANGUAGES CXX)
if(NOT APPLE)
  message(FATAL_ERROR "The native USB adapter is currently macOS-only")
endif()
find_package(JNI REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(USB REQUIRED IMPORTED_TARGET libusb-1.0>=1.0.25)
add_library(carplay_usb SHARED usb.cpp utun.cpp)
target_compile_features(carplay_usb PRIVATE cxx_std_20)
target_compile_options(carplay_usb PRIVATE -Wall -Wextra -Werror)
target_include_directories(carplay_usb PRIVATE ${JNI_INCLUDE_DIRS})
target_link_libraries(carplay_usb PRIVATE PkgConfig::USB)
add_executable(carplay_usb_descriptors descriptors_test.cpp)
target_compile_features(carplay_usb_descriptors PRIVATE cxx_std_20)
target_compile_options(carplay_usb_descriptors PRIVATE -Wall -Wextra -Werror)
target_link_libraries(carplay_usb_descriptors PRIVATE PkgConfig::USB)
