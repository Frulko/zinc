set(ZN_UV_SOURCES "")
set(ZN_UV_SECTION "")
foreach(line IN LISTS ZN_UV_LINES)
  if(line MATCHES "^#" OR line STREQUAL "")
    continue()
  elseif(line MATCHES "^\\[(.*)\\]$")
    set(ZN_UV_SECTION "${CMAKE_MATCH_1}")
  elseif(ZN_UV_SECTION STREQUAL "common" OR ZN_UV_SECTION STREQUAL "unix" OR (ZN_UV_SECTION STREQUAL "darwin" AND APPLE) OR (ZN_UV_SECTION STREQUAL "linux" AND CMAKE_SYSTEM_NAME STREQUAL "Linux"))
    list(APPEND ZN_UV_SOURCES third_party/libuv/${line})
  endif()
endforeach()

# The graphics host (ZN-027): the existing runtime (runtime/: zrt, raster, gfx) behind the zn::host::Gfx table, headless.
# Built with the runtime's own flags so its pixels match the old builds. Off where the runtime does not build.

  set(ZN_PLATFORM_NAME linux)
else()
  set(ZN_PLATFORM_NAME macos)
endif()
option(ZN_HOST_GFX "link the graphics host (zinc:gfx)" ON)
set(ZN_RUNTIME_DIR ${CMAKE_SOURCE_DIR}/../runtime)

  add_library(zn_host_gfx STATIC src/host/loop.cpp src/host/osc.cpp src/host/mqtt.cpp src/host/http.cpp src/host/tls.cpp src/host/crypto.cpp src/host/regexp.cpp src/host/sock.cpp src/host/sys_host.cpp src/host/gfx_host.cpp src/host/render_bench.cpp src/host/backend_sw.cpp src/host/resources.cpp
    ${ZN_RUNTIME_DIR}/zrt.cpp ${ZN_RUNTIME_DIR}/host.cpp ${ZN_RUNTIME_DIR}/raster.cpp ${ZN_RUNTIME_DIR}/ttf.cpp ${ZN_RUNTIME_DIR}/gfx.cpp
    src/host/hal_dispatch.cpp ${CMAKE_SOURCE_DIR}/../targets/null/hal_null.cpp ${CMAKE_SOURCE_DIR}/../targets/common/hal_posix.cpp)
  # the headless and the window HAL live in one binary: their hal_* functions are renamed, src/host/hal_dispatch.cpp picks one at run time
  set(ZN_HAL_FUNCS init shutdown frame_begin frame_end poll_input present surface_size fixed_dt)
  set(ZN_NULL_DEFS "")
  foreach(f ${ZN_HAL_FUNCS})
    list(APPEND ZN_NULL_DEFS "hal_${f}=null_hal_${f}")
  endforeach()
  set_source_files_properties(${CMAKE_SOURCE_DIR}/../targets/null/hal_null.cpp PROPERTIES COMPILE_DEFINITIONS "${ZN_NULL_DEFS}")
  # SDL3 is built from third_party/SDL3 (3.4.16) and linked statically: no dylib to ship (ZN-118). -DZN_SDL_VENDORED=OFF uses an installed SDL3 instead.
  option(ZN_SDL_VENDORED "build the vendored SDL3 statically" ON)
  if(ZN_SDL_VENDORED AND EXISTS ${CMAKE_SOURCE_DIR}/third_party/SDL3/CMakeLists.txt)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TESTS OFF CACHE BOOL "" FORCE)
    set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
    set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
    set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
    foreach(sub AUDIO CAMERA GPU HAPTIC HIDAPI JOYSTICK SENSOR POWER DIALOG TRAY VULKAN)   # what the HAL does not use: video, events, clipboard, touch, pen, GL
      set(SDL_${sub} OFF CACHE BOOL "" FORCE)
    endforeach()
    add_subdirectory(third_party/SDL3 EXCLUDE_FROM_ALL)
    target_compile_options(SDL3-static PRIVATE -w)   # vendored code: its warnings are not ours
    set(SDL3_FOUND TRUE)
    set(ZN_SDL_VENDORED_BUILD TRUE)
  else()
    if(NOT CMAKE_CROSSCOMPILING)   # a cross build (tools/cross-libs) never links the SDL3 of the machine it runs on
      find_package(SDL3 CONFIG QUIET)
    endif()
  endif()
  if(SDL3_FOUND)
    set(ZN_SDL_FUNCS init shutdown frame_begin frame_end poll_input present surface_size fixed_dt text_input clipboard_get clipboard_set set_cursor escape_by_app escape pixel_scale window_handle)
    set(ZN_SDL_DEFS "")
    foreach(f ${ZN_SDL_FUNCS})
      list(APPEND ZN_SDL_DEFS "hal_${f}=sdl_hal_${f}")
    endforeach()
    target_sources(zn_host_gfx PRIVATE ${CMAKE_SOURCE_DIR}/../targets/macos/hal_sdl.cpp)
    if(APPLE)   # the Cocoa piece of the transparent window (ZN-249)
      target_sources(zn_host_gfx PRIVATE ${CMAKE_SOURCE_DIR}/../targets/macos/hal_cocoa.mm)
      set_source_files_properties(${CMAKE_SOURCE_DIR}/../targets/macos/hal_cocoa.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc;-w")
    endif()
    set_source_files_properties(${CMAKE_SOURCE_DIR}/../targets/macos/hal_sdl.cpp PROPERTIES COMPILE_DEFINITIONS "${ZN_SDL_DEFS}")
    target_compile_definitions(zn_host_gfx PRIVATE ZN_HOST_SDL)
    target_link_libraries(zn_host_gfx PUBLIC SDL3::SDL3)
    if(ZN_SDL_VENDORED_BUILD)
      set(ZN_SDL_LIB "${CMAKE_BINARY_DIR}/third_party/SDL3/libSDL3.a")
      set(ZN_SDL_INC "${CMAKE_SOURCE_DIR}/third_party/SDL3/include")
    else()
      get_target_property(ZN_SDL_LIB SDL3::SDL3 LOCATION)
      get_target_property(ZN_SDL_INC SDL3::SDL3 INTERFACE_INCLUDE_DIRECTORIES)
    endif()
    message(STATUS "zinc: window support through SDL3 (${ZN_SDL_LIB})")
  else()
    message(STATUS "zinc: SDL3 not found, apps run headless only")
  endif()
  target_link_libraries(zn_host_gfx PUBLIC zn_uv zn_llhttp zn_mbedtls zn_regexp zn_codec)
  if(NOT ZN_TLS)
    target_compile_definitions(zn_host_gfx PRIVATE ZN_NO_TLS)
  endif()
  set_target_properties(zn_host_gfx PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
  target_include_directories(zn_host_gfx PUBLIC include PRIVATE ${ZN_RUNTIME_DIR} ${ZN_RUNTIME_DIR}/include)
  target_compile_options(zn_host_gfx PRIVATE -fno-exceptions -fno-rtti -fwrapv -ffp-contract=off -fno-threadsafe-statics -O2 -w)
  target_compile_definitions(zn_host_gfx PRIVATE ZRT_HEAP_BYTES=536870912u ZRT_PLATFORM="${ZN_PLATFORM_NAME}" ZRT_POINT_POOL=262144 ZRT_GROW_DRAW_CMDS)
endif()

  function(zn_plugin name Name)
    set(gen ${CMAKE_BINARY_DIR}/gen/${name})
    set(spec ${ZN_PLUGIN_DIR}/${name}/native/${name}.spec.ts)
    add_custom_command(OUTPUT ${gen}/zinc_native_${Name}.h ${gen}/zinc_native_${Name}_thunk.cpp
      COMMAND ${CMAKE_COMMAND} -E make_directory ${gen}
      COMMAND zinc native-gen ${spec} ${gen}
      COMMAND zinc native-gen --thunk ${spec} ${gen}
      DEPENDS zinc ${spec} VERBATIM)
    add_library(zn_plugin_${name} STATIC ${ZN_PLUGIN_DIR}/${name}/native/${name}.host.cpp ${gen}/zinc_native_${Name}_thunk.cpp ${ARGN})
    target_include_directories(zn_plugin_${name} PRIVATE ${gen} ${CMAKE_SOURCE_DIR}/src PUBLIC ${CMAKE_SOURCE_DIR}/include ${ZN_RUNTIME_DIR} ${ZN_RUNTIME_DIR}/include)
    set_target_properties(zn_plugin_${name} PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
    target_compile_options(zn_plugin_${name} PRIVATE -fno-exceptions -fno-rtti -fwrapv -ffp-contract=off -fno-threadsafe-statics -O2 -w)
    target_compile_definitions(zn_plugin_${name} PRIVATE ZRT_HEAP_BYTES=536870912u ZRT_PLATFORM="macos" ZRT_POINT_POOL=262144 ZRT_GROW_DRAW_CMDS)
  endfunction()
