cmake_minimum_required(VERSION 3.20)
# Set these before compiler detection: all archives must use the same deployment target.
set(CMAKE_OSX_ARCHITECTURES x86_64 CACHE STRING "legacy Mac architecture" FORCE)
set(CMAKE_OSX_DEPLOYMENT_TARGET 10.12 CACHE STRING "legacy Mac deployment target" FORCE)
project(zinc_macos_legacy C CXX OBJCXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Darwin")
  message(FATAL_ERROR "The macOS legacy runtime requires a macOS SDK")
endif()
set(CMAKE_OBJCXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
set(ZN_CONSOLE macos CACHE STRING "runtime platform" FORCE)
set(ZN_CONSOLE_HAL "${CMAKE_CURRENT_SOURCE_DIR}/hal_cocoa.mm" CACHE FILEPATH "legacy window HAL" FORCE)
add_subdirectory(../console core)
target_sources(zinc-core PRIVATE ../common/hal_posix.cpp)
target_compile_definitions(zinc-core PRIVATE ZN_CORE_EXIT_ON_FINISH)
target_link_libraries(zinc-core PRIVATE "-framework Cocoa")
set_target_properties(zinc-core PROPERTIES OUTPUT_NAME zinc-legacy RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
